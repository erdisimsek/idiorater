Google Apps Script Code

1. Open script.google.com > New project.
2. Paste the code below > Click Deploy > New deployment > Select type: Web App.
3. Set Execute as: Me, Who has access: Anyone > Copy Web App URL. 

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var folderName = "Live_Idiodynamic_Research_Data";
    var folders = DriveApp.getFoldersByName(folderName);
    var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
    
    var timePrefix = Utilities.formatDate(new Date(), "GMT+3", "yyyyMMdd_HHmmss");
    var baseName = timePrefix + "_" + (data.participantId || "Anon") + "_" + (data.dimension || "Rating");
    
    // Save CSV
    var csvFile = folder.createFile(baseName + ".csv", data.csvContent, MimeType.CSV);
    
    // Save PNG Graph (if provided)
    if (data.graphBase64) {
      var decodedImage = Utilities.base64Decode(data.graphBase64.replace(/^data:image\/(png|jpeg);base64,/, ""));
      var blob = Utilities.newBlob(decodedImage, MimeType.PNG, baseName + "_Graph.png");
      folder.createFile(blob);
    }
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      folder: folder.getName(),
      csvFile: csvFile.getUrl()
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
